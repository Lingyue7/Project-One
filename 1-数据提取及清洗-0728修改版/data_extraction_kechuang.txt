proc sql;
connect to greenplm as mppconn(
    dsn=warehouse310
    authdomain=gppaaauth
);

create table kechuang_potential_cust_final as
select * from connection to mppconn(
WITH
/* ① 科创人才表 */
kechuang_base AS (
    SELECT DISTINCT
        k."客户编号"          AS CST_ID,
        k."当前aum"           AS CUR_AUM,
        k."当前lum"           AS CUR_LUM,
        k."科技人才对应得分"  AS TECH_TALENT_SCORE,
        k."kum分"             AS KUM_SCORE,
        k."lum分"             AS LUM_SCORE,
        k."aum分"             AS AUM_SCORE,
        k."总分"              AS TOTAL_SCORE,
        k."档位"              AS TIER
    FROM Ngb310view.sh_yyw_qlmd_tojk006 k
),

/* ② 消贷账户全量：保留截至观察日（2026-07-27）已经生效的贷款；
   尚未到期的贷款也保留，其基础Y终点快照日将在后续截断至观察日 */
loan_all AS (
    SELECT DISTINCT
        L.P6_CUST_NUM                                        AS CST_ID,
        '31001' || L.RT_CTL3 || L.RT_CTL4 || L.RT_ACCT_NUM AS LOANACCTNO,
        L.RT_ACCT_EFF_DATE_1                                 AS ACCT_EFF_DT,
        L.RT_CURR_MATUR_DATE_1                               AS MATUR_DT
    FROM Ngb310view.T209_LS_AMWKPL08_H L
    WHERE SUBSTR(L.AMWKPL08_FILLER,1,4) IN (
              '0063','0163','0045','0046','0064',
              '0065','0246','0217','0218','0219','0220'
          )
      AND L.RT_ACCT_EFF_DATE_1 IS NOT NULL
      AND L.RT_CURR_MATUR_DATE_1 IS NOT NULL
      AND L.RT_ACCT_EFF_DATE_1 >= DATE'2025-01-01'
      AND L.RT_ACCT_EFF_DATE_1 <= DATE'2026-07-27'
),

/* ③ 科创 ∩ 消贷（每笔贷款一行） */
intersection_base AS (
    SELECT
        kb.CST_ID,
        kb.CUR_AUM,
        kb.CUR_LUM,
        kb.TECH_TALENT_SCORE,
        kb.KUM_SCORE,
        kb.LUM_SCORE,
        kb.AUM_SCORE,
        kb.TOTAL_SCORE,
        kb.TIER,
        la.LOANACCTNO,
        la.ACCT_EFF_DT,
        la.MATUR_DT,
        /* 基础Y终点快照日：贷款到期日与观察日中较早的一天 */
        LEAST(
            la.MATUR_DT,
            DATE'2026-07-27'
        ) AS END_SNAP_DT,
        /* X特征时点：贷款生效日上一个月的月末 */
        (DATE_TRUNC('month', la.ACCT_EFF_DT) - INTERVAL '1 day')::date AS X_SNAP_DT
    FROM kechuang_base kb
    INNER JOIN loan_all la
        ON kb.CST_ID = la.CST_ID
),

/* ④ 授信台账（有效记录） */
credit_base AS (
    SELECT DISTINCT
        C.LOANACCTNO,
        C.CREDAMT,
        C.ACCTBEGINDATE,
        C.BUSIKIND
    FROM Ngb310view.DIM_LM_CRG_STDGBOOK_INF_H C
    WHERE C.EDDT = '2999-12-31'
),

/* ⑤ 贷款开始时点快照：保留所有起点账户状态，不筛选起点是否违约 */
loan_snap_start AS (
    SELECT
        '31001' || L.RT_CTL3 || L.RT_CTL4 || L.RT_ACCT_NUM AS LOANACCTNO,
        L.AC_CURR_BAL                AS AC_CURR_BAL_START,
        L.BA_OUT_BAL                 AS BA_OUT_BAL_START,
        L.RT_ACCT_STAT_2             AS RT_ACCT_STAT_2_START
    FROM Ngb310view.T209_LS_AMWKPL08_H L
    INNER JOIN intersection_base ib
        ON '31001' || L.RT_CTL3 || L.RT_CTL4 || L.RT_ACCT_NUM = ib.LOANACCTNO
       AND L.STDT <= ib.ACCT_EFF_DT
       AND L.EDDT  >  ib.ACCT_EFF_DT
),

/* ⑥ 频繁支用因变量终点快照：取基础Y终点快照日当天余额。
   仅提取参与该任务的两个余额字段，减少远程表读取列数。 */
loan_snap_balance_end AS (
    SELECT
        '31001' || L.RT_CTL3 || L.RT_CTL4 || L.RT_ACCT_NUM AS LOANACCTNO,
        L.AC_CURR_BAL                AS AC_CURR_BAL_END,
        L.BA_OUT_BAL                 AS BA_OUT_BAL_END
    FROM Ngb310view.T209_LS_AMWKPL08_H L
    INNER JOIN intersection_base ib
        ON '31001' || L.RT_CTL3 || L.RT_CTL4 || L.RT_ACCT_NUM = ib.LOANACCTNO
       AND L.STDT <= ib.END_SNAP_DT
       AND L.EDDT  >  ib.END_SNAP_DT
),

/* ⑦ X特征：对私客户基础信息（固定11Jun2026快照） */
cust_feature AS (
    SELECT
        C.CST_ID,
        C.CST_BLNG_INSID,
        C.AGE,
        C.GND_CD,
        C.MAR_STTN_CD,
        C.EDUCATION_CD,
        C.OCCUP_CD,
        C.BANK_CUST_BECOME_DATE,
        C.CST_STAR_CD
    FROM Ngb310view.TAG_DMG_CUST_INFO C
    WHERE C.DATA_DT = '11Jun2026'
)

/* ⑨ 最终拼接 */
SELECT

    IB.LOANACCTNO,
    IB.ACCT_EFF_DT,
    IB.MATUR_DT,
    IB.END_SNAP_DT,
    IB.X_SNAP_DT,

    /* 科创标签 */
    IB.CUR_AUM,
    IB.CUR_LUM,
    IB.TECH_TALENT_SCORE,
    IB.KUM_SCORE,
    IB.LUM_SCORE,
    IB.AUM_SCORE,
    IB.TOTAL_SCORE,

    /* 授信台账 */

    LS_E.BA_OUT_BAL_END     - LS_S.BA_OUT_BAL_START     AS BA_OUT_BAL_DIFF,

    /* 贷款起点、终点状态及余额校验值 */
    LS_S.RT_ACCT_STAT_2_START,
    LS_S.AC_CURR_BAL_START,
    LS_E.AC_CURR_BAL_END,
    LS_S.BA_OUT_BAL_START,
    LS_E.BA_OUT_BAL_END,

    /* X：对私客户基础信息 */
    CF.CST_BLNG_INSID,
    CF.AGE,
    CF.GND_CD,
    CF.MAR_STTN_CD,
    CF.EDUCATION_CD,
    CF.OCCUP_CD,
    CF.BANK_CUST_BECOME_DATE,
    CF.CST_STAR_CD,

    /* X：近十二个月消费行为（按字段截图逐项核对后的86个字段） */
    PF.PRVTCSTRT12MOCNS1PAMT,
    PF.PRVTCSTRT12MO_CSMDNUM,
    PF.PRVTCSTRT12M1CNS1PAMT,
    PF.PRVTCSTRT12MOACSMDNUM,
    PF.PRVTCSTR12MGHCCNMPAMT,

    PF.PRVTCSTR12MJTCCNMPAMT,

    PF.PRVTCSTR12MLAMCCNPAMT,
    PF.PRVTCSTR12MLAMCCMDNUM,
    PF.PRVTCSTR12MPHCCNMPAMT,

    PF.PRVTCSTR12MCWSSCCPAMT,
    PF.PRVTCSTR12MCWSSCCDNUM,
    PF.PRVTCSTR12MGSCCNMPAMT,
    PF.PRVTCSTR12MGSCCSMDNUM,
    PF.PRVTCSTR12MGMCCNMPAMT,
    PF.PRVTCSTR12MGMCCSMDNUM,
    PF.PRVTCSTR12MCCCNSMPAMT,

LEFT JOIN credit_base     CR
       ON IB.LOANACCTNO = CR.LOANACCTNO
LEFT JOIN loan_snap_start LS_S
       ON IB.LOANACCTNO = LS_S.LOANACCTNO
LEFT JOIN loan_snap_balance_end LS_E
       ON IB.LOANACCTNO = LS_E.LOANACCTNO
LEFT JOIN cust_feature    CF
       ON IB.CST_ID = CF.CST_ID
LEFT JOIN Ngb310view.T209_FIDI_PRVT_CST_BEHAVI_INF PF
       ON PF.CST_ID  = IB.CST_ID
      AND PF.DATA_DT = IB.X_SNAP_DT
LEFT JOIN Ngb310view.T209_FIDI_PRVT_CST_MPB_STATINF MF
       ON MF.CST_ID  = IB.CST_ID
      AND MF.DATA_DT = IB.X_SNAP_DT
