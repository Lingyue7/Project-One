/*
科创人才与消贷客户交集一体化提取：基础特征 + 频繁支用任务数据 + 违约风险任务数据。
仅执行一个 SAS PROC SQL 段，输出一张表 kechuang_loan_all_final，无需另行合并风险结果。
观察日：2026-09-04。以消贷合并版为基础，仅增加人才名单交集及人才特征。
按 CST_ID 内连接科创人才名单，提取原名单的 8 个特征。
人才名单 DISTINCT 沿用原脚本；同一客户存在不同人才特征记录时仍可能输出多行。
贷款生效日范围：2025-01-01 至 2026-09-04（含）；保留原消贷产品代码筛选。
余额终点 END_SNAP_DT = min(贷款到期日, 观察日)。
违约与频繁支用共用 END_SNAP_DT，不再减一天。
违约任务提取 T209_PM_RSKCL_CRNRSLT_H.SEQUOVERDAY（零售客户贷款连续拖欠天数）。
仅提取原始天数，不在本 SQL 中设置违约阈值；缺失值保留为空。
风险表使用 STDT / EDDT 选取历史快照；沿用原脚本左闭右开口径：STDT <= END_SNAP_DT < EDDT。
ACCTBEGINDATE / ENDDATE 分别为贷款开户日期、到期日期，不作为历史记录有效期条件。
消费、手机银行特征仍取贷款生效前月末；客户维表仅限制 EDDT < '2999-12-31'，仅按 CST_ID 关联；
授信台账仍取 EDDT = '2999-12-31' 的记录。
客户维表在所选字段组合上使用 DISTINCT 去重；若同一 CST_ID 的字段取值不同，仍会保留多条记录并可能放大贷款行数。
余额快照按每行账户和起终点日期直接关联；源表有效期重叠等情况仍可能导致重复行。
未连接数据库执行验证。
*/

proc sql;
connect to greenplm as mppconn(
    dsn=warehouse310
    authdomain=gppaaauth
);

create table kechuang_loan_all_final as
select * from connection to mppconn(
WITH
/* ① 科创人才名单及原有特征 */
kechuang_base AS (
    SELECT DISTINCT
        k."客户编号"          AS CST_ID,
        k."当前aum"           AS CUR_AUM,
        k."当前lum"           AS CUR_LUM,
        k."科技人才对应得分"  AS TECH_TALENT_SCORE,
        k."kum分"             AS KUM_SCORE,
        k."lum分"             AS LUM_SCORE,
        k."aum分"             AS AUM_SCORE,
        k."总分"              AS TOTAL_SCORE,
        k."档位"              AS TIER
    FROM Ngb310view.sh_yyw_qlmd_tojk006 k
),

/* ② 消贷账户全量：保留截至观察日（2026-09-04）已经生效的贷款；
   尚未到期的贷款也保留，其基础Y终点快照日将在后续截断至观察日 */
loan_all AS (
    SELECT DISTINCT
        L.P6_CUST_NUM                                        AS CST_ID,
        '31001' || L.RT_CTL3 || L.RT_CTL4 || L.RT_ACCT_NUM AS LOANACCTNO,
        L.RT_ACCT_EFF_DATE_1                                 AS ACCT_EFF_DT,
        L.RT_CURR_MATUR_DATE_1                               AS MATUR_DT
    FROM Ngb310view.T209_LS_AMWKPL08_H L
    WHERE SUBSTR(L.AMWKPL08_FILLER,1,4) IN (
              '0063','0163','0045','0046','0064',
              '0065','0246','0217','0218','0219','0220'
          )
      AND L.RT_ACCT_EFF_DATE_1 IS NOT NULL
      AND L.RT_CURR_MATUR_DATE_1 IS NOT NULL
      AND L.RT_ACCT_EFF_DATE_1 >= DATE'2025-01-01'
      AND L.RT_ACCT_EFF_DATE_1 <= DATE'2026-09-04'
),

/* ③ 科创人才与消贷账户交集 */
consumer_loan_base AS (
    SELECT
        la.CST_ID,
        KB.CUR_AUM,
        KB.CUR_LUM,
        KB.TECH_TALENT_SCORE,
        KB.KUM_SCORE,
        KB.LUM_SCORE,
        KB.AUM_SCORE,
        KB.TOTAL_SCORE,
        KB.TIER,
        la.LOANACCTNO,
        la.ACCT_EFF_DT,
        la.MATUR_DT,
        /* 基础Y终点快照日：贷款到期日与观察日中较早的一天 */
        LEAST(
            la.MATUR_DT,
            DATE'2026-09-04'
        ) AS END_SNAP_DT,
        /* X特征时点：贷款生效日上一个月的月末 */
        (DATE_TRUNC('month', la.ACCT_EFF_DT) - INTERVAL '1 day')::date AS X_SNAP_DT
    FROM loan_all la
    INNER JOIN kechuang_base KB ON la.CST_ID = KB.CST_ID
),

/* ④ 授信台账（有效记录） */
credit_base AS (
    SELECT DISTINCT
        C.LOANACCTNO,
        C.CREDAMT,
        C.ACCTBEGINDATE,
        C.BUSIKIND
    FROM Ngb310view.DIM_LM_CRG_STDGBOOK_INF_H C
    WHERE C.EDDT = '2999-12-31'
),

/* ⑦ X特征：对私客户维表，仅按指定 EDDT 条件筛选，不限制其他时间 */
cust_feature AS (
    SELECT DISTINCT
        C.CST_ID,
        C.BRTH_DT,
        C.GND_CD,
        C.MAR_STTN_CD,
        C.EDDGR_CD,
        C.OCP_CD,
        C.CST_STAR_CD
    FROM Ngb310view.DIM_IDV_CST_H C
    WHERE C.EDDT < '2999-12-31'
)

/* ⑨ 最终拼接 */
SELECT
    IB.CST_ID,
    IB.LOANACCTNO,
    IB.ACCT_EFF_DT,
    IB.MATUR_DT,
    IB.END_SNAP_DT,
    IB.X_SNAP_DT,

    /* 科创人才特征 */
    IB.CUR_AUM,
    IB.CUR_LUM,
    IB.TECH_TALENT_SCORE,
    IB.KUM_SCORE,
    IB.LUM_SCORE,
    IB.AUM_SCORE,
    IB.TOTAL_SCORE,
    IB.TIER,

    /* 违约任务：基础Y终点快照日的连续拖欠天数 */
    R.SEQUOVERDAY,

    /* 授信台账 */
    CR.CREDAMT,
    CR.ACCTBEGINDATE,
    CR.BUSIKIND,

    /* Y差值 */
    LS_E.AC_CURR_BAL    - LS_S.AC_CURR_BAL    AS AC_CURR_BAL_DIFF,
    LS_E.BA_OUT_BAL     - LS_S.BA_OUT_BAL     AS BA_OUT_BAL_DIFF,

    /* 贷款起点、终点余额校验值 */
    LS_S.AC_CURR_BAL AS AC_CURR_BAL_START,
    LS_E.AC_CURR_BAL AS AC_CURR_BAL_END,
    LS_S.BA_OUT_BAL AS BA_OUT_BAL_START,
    LS_E.BA_OUT_BAL AS BA_OUT_BAL_END,

    /* X：对私客户基础信息 */
    CF.BRTH_DT,
    CF.GND_CD,
    CF.MAR_STTN_CD,
    CF.EDDGR_CD,
    CF.OCP_CD,
    CF.CST_STAR_CD,

    /* X：近十二个月消费行为（按字段截图逐项核对后的86个字段） */
    PF.PRVTCSTRT12MOCNS1PAMT,
    PF.PRVTCSTRT12MO_CSMDNUM,
    PF.PRVTCSTRT12M1CNS1PAMT,
    PF.PRVTCSTRT12MOACSMDNUM,

    PF.PRVTCSTR12MCCCNS1PAMT,
    PF.PRVTCSTRT12MCCCS1DNUM,
    PF.PRVTCSTR12MJTCCNMPAMT,
    PF.PRVTCSTR12MJTCCSMDNUM,
    PF.PRVTCSTR12MECCNSMPAMT,

    PF.PRVTCSTR12MRECCNMPAMT,
    PF.PRVTCSTR12MRECCSMDNUM,
    PF.PRVTCSTR12MCSCCN1PAMT,
    PF.PRVTCSTR12MCSCCS1DNUM,
    PF.PRVTCSTR12MWCCNSMPAMT,
    PF.PRVTCSTRT12MWCCSMDNUM,
    PF.PRVTCSTR12MATCCNMPAMT,
    PF.PRVTCSTR12MATCCSMDNUM,

    PF.PRVTCSTR12MSCCNSMPAMT,

    PF.PRVTCSTRT12MRCCS1DNUM,
    PF.PRVTCSTR12MLAMCCNPAMT,
    PF.PRVTCSTR12MLAMCCMDNUM,
    PF.PRVTCSTR12MPHCCNMPAMT,
    PF.PRVTCSTR12MPHCCSMDNUM,
    PF.PRVTCSTR12MPSCCNMPAMT,
    PF.PRVTCSTR12MPSCCSMDNUM,
    PF.PRVTCSTR12MCWSSCCPAMT,
    PF.PRVTCSTR12MCWSSCCDNUM,
    PF.PRVTCSTR12MGSCCNMPAMT,
    PF.PRVTCSTR12MGSCCSMDNUM,
    PF.PRVTCSTR12MGMCCNMPAMT,
    PF.PRVTCSTR12MGMCCSMDNUM,
    PF.PRVTCSTR12MCCCNSMPAMT,
    PF.PRVTCSTRT12MCCCSMDNUM,
    PF.PRVTCSTR12MDCCNSMPAMT,
    PF.PRVTCSTRT12MDCCSMDNUM,
    PF.PRVTCSTR12MGCCNSMPAMT,
    PF.PRVTCSTRT12MGCCS1DNUM,

    MF.LBLCLBRMPBACTVCST_IND,

LEFT JOIN credit_base     CR
       ON IB.LOANACCTNO = CR.LOANACCTNO
/* 余额按本行账户与日期直接匹配，避免样本先展开再回连。 */
LEFT JOIN Ngb310view.T209_LS_AMWKPL08_H LS_S
       ON '31001' || LS_S.RT_CTL3 || LS_S.RT_CTL4 || LS_S.RT_ACCT_NUM = IB.LOANACCTNO
      AND LS_S.STDT <= IB.ACCT_EFF_DT
      AND LS_S.EDDT  > IB.ACCT_EFF_DT
LEFT JOIN Ngb310view.T209_LS_AMWKPL08_H LS_E
       ON '31001' || LS_E.RT_CTL3 || LS_E.RT_CTL4 || LS_E.RT_ACCT_NUM = IB.LOANACCTNO
      AND LS_E.STDT <= IB.END_SNAP_DT
      AND LS_E.EDDT  > IB.END_SNAP_DT
/* 风险快照直接按账户和终点日期关联，不新增排序、聚合或样本回连。 */
LEFT JOIN Ngb310view.T209_PM_RSKCL_CRNRSLT_H R
       ON R.LOANACCTNO = IB.LOANACCTNO
      AND R.STDT <= IB.END_SNAP_DT
      AND R.EDDT  > IB.END_SNAP_DT
LEFT JOIN cust_feature    CF
       ON IB.CST_ID = CF.CST_ID
LEFT JOIN Ngb310view.T209_FIDI_PRVT_CST_BEHAVI_INF PF
       ON PF.CST_ID  = IB.CST_ID
      AND PF.DATA_DT = IB.X_SNAP_DT
LEFT JOIN Ngb310view.T209_FIDI_PRVT_CST_MPB_STATINF MF
       ON MF.CST_ID  = IB.CST_ID
      AND MF.DATA_DT = IB.X_SNAP_DT
);

quit;
